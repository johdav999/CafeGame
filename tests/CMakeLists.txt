cmake_minimum_required(VERSION 3.24)
project(CafeSimulationTests LANGUAGES CXX)
include(CTest)
add_executable(cafe_garden_story garden_story_test.cpp)
target_include_directories(cafe_garden_story PRIVATE ../src)
target_link_libraries(cafe_garden_story PRIVATE WebEngine::Simulator)
if(EMSCRIPTEN)
  set_target_properties(cafe_garden_story PROPERTIES SUFFIX ".cjs")
  target_compile_definitions(cafe_garden_story PRIVATE GARDEN_G4_FIXTURE="/garden-g4.bin" GARDEN_BASE_FIXTURE="/garden-base.bin")
  target_link_options(cafe_garden_story PRIVATE -sENVIRONMENT=node -sEXIT_RUNTIME=1 -sALLOW_MEMORY_GROWTH=1 -sSTACK_SIZE=2097152
    "SHELL:--embed-file ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/garden-g4-campaign23.bin@/garden-g4.bin"
    "SHELL:--embed-file ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/garden-g1-base19.bin@/garden-base.bin")
else()
  target_compile_definitions(cafe_garden_story PRIVATE GARDEN_G4_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/fixtures/garden-g4-campaign23.bin" GARDEN_BASE_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/fixtures/garden-g1-base19.bin")
endif()
add_test(NAME cafe_garden_story COMMAND ${CMAKE_CROSSCOMPILING_EMULATOR} $<TARGET_FILE:cafe_garden_story>)
